The 5 Endpoints Explained
1. GET /fhir/r4/Patient/{id}
Client asks "give me patient 123." Code looks up that ID in the in-memory patients dict. Found → return it. Not found → HTTP 404. Most common EHR read operation and the most common target for unauthorized access attacks.

2. POST /fhir/r4/MedicationRequest
Client sends JSON with drug name, patient ID, dosage. Code generates a new ID, timestamps it, appends to medications list. Simulates prescribing — targeted for unauthorized prescription creation attacks.

3. GET /fhir/r4/Observation
Returns all lab results from the observations list. Supports optional ?subject=Patient/p001 query filter. Targeted for bulk data exfiltration attacks.

4. POST /fhir/r4/Appointment
Client sends patient ID + datetime + reason. Code creates an appointment record, appends to appointments list. Targeted for patient-record tampering attacks.

5. GET /fhir/r4/metadata ← Attack Vector #3
Returns a CapabilityStatement — a FHIR-standard JSON that lists every resource and operation your server supports. In real EHRs this is public and unauthenticated. Attackers hit this first to map the entire attack surface. Your mock must return this accurately — it's what makes the simulation realistic.

@app.get("/fhir/r4/Observation")
def get_observations(subject: str = None):
    if subject:
        filtered = [o for o in observations if o["subject"]["reference"] == subject]
    else:
        filtered = observations
    return {
        "resourceType": "Bundle",
        "type": "searchset",
        "total": len(filtered),
        "entry": [{"resource": o} for o in filtered]
    }

import sqlite3
import json
import os

# Paths
DB_PATH = "fhir.db"
SYNTHEA_FOLDER = "synthea_sample_data_fhir_latest"

# Connect to SQLite (creates fhir.db if it doesn't exist)
conn = sqlite3.connect(DB_PATH)
cursor = conn.cursor()

# Create tables
cursor.execute("""
    CREATE TABLE IF NOT EXISTS patients (
        id          TEXT PRIMARY KEY,
        name        TEXT,
        gender      TEXT,
        birth_date  TEXT
    )
""")

cursor.execute("""
    CREATE TABLE IF NOT EXISTS observations (
        id      TEXT PRIMARY KEY,
        subject TEXT,
        code    TEXT,
        status  TEXT
    )
""")

cursor.execute("""
    CREATE TABLE IF NOT EXISTS medication_requests (
        id          TEXT PRIMARY KEY,
        subject     TEXT,
        medication  TEXT,
        status      TEXT,
        authored_on TEXT
    )
""")

# Counters
patients_loaded = 0
observations_loaded = 0
medications_loaded = 0

# Loop through every file in the Synthea folder
for filename in os.listdir(SYNTHEA_FOLDER):

    # Skip hospital and practitioner files
    if filename.startswith("hospital") or filename.startswith("practitioner"):
        continue

    filepath = os.path.join(SYNTHEA_FOLDER, filename)

    try:
        with open(filepath, "r", encoding="utf-8") as f:
            bundle = json.load(f)

        # Loop through every entry in the Bundle
        for entry in bundle.get("entry", []):
            resource = entry.get("resource", {})
            resource_type = resource.get("resourceType", "")

            # Patient
            if resource_type == "Patient":
                pid = resource.get("id", "")
                name_obj = resource.get("name", [{}])[0]
                given = name_obj.get("given", [""])[0]
                family = name_obj.get("family", "")
                name = given + " " + family
                gender = resource.get("gender", "unknown")
                birth_date = resource.get("birthDate", "")

                cursor.execute("""
                    INSERT OR IGNORE INTO patients (id, name, gender, birth_date)
                    VALUES (?, ?, ?, ?)
                """, (pid, name, gender, birth_date))
                patients_loaded += 1

            # Observation
            elif resource_type == "Observation":
                oid = resource.get("id", "")
                subject = resource.get("subject", {}).get("reference", "")
                code = resource.get("code", {}).get("text", "Unknown")
                status = resource.get("status", "")

                cursor.execute("""
                    INSERT OR IGNORE INTO observations (id, subject, code, status)
                    VALUES (?, ?, ?, ?)
                """, (oid, subject, code, status))
                observations_loaded += 1

            # MedicationRequest
            elif resource_type == "MedicationRequest":
                mid = resource.get("id", "")
                subject = resource.get("subject", {}).get("reference", "")
                medication = resource.get("medicationCodeableConcept", {}).get("text", "Unknown")
                status = resource.get("status", "")
                authored_on = resource.get("authoredOn", "")

                cursor.execute("""
                    INSERT OR IGNORE INTO medication_requests (id, subject, medication, status, authored_on)
                    VALUES (?, ?, ?, ?, ?)
                """, (mid, subject, medication, status, authored_on))
                medications_loaded += 1

    except Exception as e:
        print(f"Skipping {filename}: {e}")
        continue

# Save and close
conn.commit()
conn.close()

# Summary
print(f"Patients loaded:       {patients_loaded}")
print(f"Observations loaded:   {observations_loaded}")
print(f"Medications loaded:    {medications_loaded}")
print(f"Database saved to:     {DB_PATH}")
